a) 

foldr (-) 0 [0, 5, 4, 3]
= 0 - (foldr (-) 0 [5, 4, 3])
= 0 - (5 - (foldr (-) 0 [4, 3]))
= 0 - (5 - (4 - (foldr (-) 0 [3])))
= 0 - (5 - (4 - (3 - (foldr (-) 0 []))))
= 0 - (5 - (4 - (3 - 0)))
= 0 - (5 - (4 - 3))
= 0 - (5 - 1)
= 0 - 4
= -4

= foldl (-) (0 - 5) [4, 3]
= foldl (-) (-5) [4, 3]
= foldl (-) ((-5) - 4) [3]
